cmake_minimum_required(VERSION 3.22)
project(JerzyMasterLabX VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MASTERLAB_PLUGIN "Build JUCE VST3" ON)
add_library(masterlab INTERFACE)
target_include_directories(masterlab INTERFACE Source)
add_executable(dsp_benchmark tests/Benchmark.cpp)
target_link_libraries(dsp_benchmark PRIVATE masterlab)
enable_testing()
add_executable(dsp_tests tests/DspTests.cpp)
target_link_libraries(dsp_tests PRIVATE masterlab)
add_test(NAME dsp COMMAND dsp_tests)
add_executable(routing_tests tests/RoutingTests.cpp)
target_link_libraries(routing_tests PRIVATE masterlab)
add_test(NAME routing COMMAND routing_tests)
if(MASTERLAB_PLUGIN)
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
 juce_add_plugin(JerzyMasterLabX BUNDLE_ID com.jerzymasterlab.masterlabx COMPANY_NAME "Jerzy MasterLab" PLUGIN_MANUFACTURER_CODE Jrzy PLUGIN_CODE Jmlx FORMATS VST3 PRODUCT_NAME "JerzyMasterLabX" COPY_PLUGIN_AFTER_BUILD FALSE)
 target_sources(JerzyMasterLabX PRIVATE Source/Plugin.cpp)
 target_link_libraries(JerzyMasterLabX PRIVATE masterlab juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
 target_compile_definitions(JerzyMasterLabX PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 add_executable(plugin_tests)
 target_sources(plugin_tests PRIVATE tests/PluginTests.cpp)
 target_link_libraries(plugin_tests PRIVATE JerzyMasterLabX)
 target_include_directories(plugin_tests PRIVATE "${juce_SOURCE_DIR}/modules")
 target_compile_definitions(plugin_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
 add_test(NAME plugin_state COMMAND plugin_tests)
 add_executable(state_tests tests/StateTests.cpp)
 target_link_libraries(state_tests PRIVATE JerzyMasterLabX)
 target_include_directories(state_tests PRIVATE "${juce_SOURCE_DIR}/modules")
 target_compile_definitions(state_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
 add_test(NAME state_validation COMMAND state_tests)
endif()
