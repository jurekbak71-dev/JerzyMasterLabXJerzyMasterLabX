cmake_minimum_required(VERSION 3.22)
project(JerzyMasterLabPro VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
foreach(test rack measure layout controller gain)
  add_executable(masterlabpro_${test}_test tests/${test}.cpp)
  target_include_directories(masterlabpro_${test}_test PRIVATE src)
  if(MSVC)
    target_compile_options(masterlabpro_${test}_test PRIVATE /W4 /WX)
  else()
    target_compile_options(masterlabpro_${test}_test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME ${test} COMMAND masterlabpro_${test}_test)
endforeach()

option(MASTERLABPRO_GUI_CHECK "Compile JUCE GUI components" OFF)
option(MASTERLABPRO_CONSOLE "Build standalone development GUI" OFF)
if(MASTERLABPRO_GUI_CHECK OR MASTERLABPRO_CONSOLE)
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
if(MASTERLABPRO_GUI_CHECK)
  add_executable(masterlabpro_gui_compile tests/gui_compile.cpp)
  target_include_directories(masterlabpro_gui_compile PRIVATE src)
  target_link_libraries(masterlabpro_gui_compile PRIVATE juce::juce_audio_processors juce::juce_gui_basics)
  target_compile_definitions(masterlabpro_gui_compile PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  add_test(NAME gui_compile COMMAND masterlabpro_gui_compile)
  set(MASTERLABPRO_ASSET_ROOT "" CACHE PATH "Approved A+C assets directory for strict GUI render")
  if(WIN32)
    add_executable(masterlabpro_gui_render tests/gui_render.cpp)
    target_include_directories(masterlabpro_gui_render PRIVATE src)
    target_link_libraries(masterlabpro_gui_render PRIVATE juce::juce_audio_processors juce::juce_gui_basics)
    target_compile_definitions(masterlabpro_gui_render PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    add_test(NAME gui_render COMMAND masterlabpro_gui_render rack-smoke.png)
    if(MASTERLABPRO_ASSET_ROOT)
      add_test(NAME gui_render_ac COMMAND masterlabpro_gui_render rack-ac.png "${MASTERLABPRO_ASSET_ROOT}")
    endif()
  endif()
endif()

if(MASTERLABPRO_CONSOLE)
  juce_add_gui_app(MasterLabProConsole PRODUCT_NAME "Jerzy MasterLabPro GUI")
  target_sources(MasterLabProConsole PRIVATE src/ConsoleMain.cpp)
  target_include_directories(MasterLabProConsole PRIVATE src)
  target_link_libraries(MasterLabProConsole PRIVATE juce::juce_gui_extra juce::juce_audio_processors)
  target_compile_definitions(MasterLabProConsole PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
endif()
