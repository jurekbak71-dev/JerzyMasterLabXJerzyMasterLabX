option(MASTERLABPRO_ASAN_UBSAN "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(MASTERLABPRO_TSAN "Enable ThreadSanitizer" OFF)
if(MASTERLABPRO_ASAN_UBSAN AND MASTERLABPRO_TSAN)
  message(FATAL_ERROR "ASan/UBSan and TSan must run in separate builds")
endif()
if((MASTERLABPRO_ASAN_UBSAN OR MASTERLABPRO_TSAN) AND MSVC)
  message(FATAL_ERROR "Requested sanitizer configuration is not supported by this MSVC project")
endif()
if(MASTERLABPRO_ASAN_UBSAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(MASTERLABPRO_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME assets COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/assets.py")
foreach(test rack measure layout controller gain audio_rack spectral capture analysis_history realtime module_abi parameter_registry live_host_exchange)
  add_executable(masterlabpro_${test}_test tests/${test}.cpp)
  target_include_directories(masterlabpro_${test}_test PRIVATE src sdk)
  if(MSVC)
    target_compile_options(masterlabpro_${test}_test PRIVATE /W4 /WX /UNDEBUG)
  else()
    target_compile_options(masterlabpro_${test}_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  add_test(NAME ${test} COMMAND masterlabpro_${test}_test)
endforeach()
