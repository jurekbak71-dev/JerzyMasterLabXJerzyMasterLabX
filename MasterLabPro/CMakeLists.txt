cmake_minimum_required(VERSION 3.22)
project(JerzyMasterLabPro VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
option(MASTERLABPRO_ASAN_UBSAN "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(MASTERLABPRO_TSAN "Enable ThreadSanitizer" OFF)
if(MASTERLABPRO_ASAN_UBSAN AND MASTERLABPRO_TSAN)
  message(FATAL_ERROR "ASan/UBSan and TSan must run in separate builds")
endif()
if((MASTERLABPRO_ASAN_UBSAN OR MASTERLABPRO_TSAN) AND MSVC)
  message(FATAL_ERROR "Requested sanitizer configuration is not supported by this MSVC project")
endif()
if(MASTERLABPRO_ASAN_UBSAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(MASTERLABPRO_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME assets COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/assets.py")
foreach(test rack measure layout controller gain audio_rack spectral capture analysis_history realtime)
  add_executable(masterlabpro_${test}_test tests/${test}.cpp)
  target_include_directories(masterlabpro_${test}_test PRIVATE src)
  if(MSVC)
    target_compile_options(masterlabpro_${test}_test PRIVATE /W4 /WX)
  else()
    target_compile_options(masterlabpro_${test}_test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME ${test} COMMAND masterlabpro_${test}_test)
endforeach()
add_executable(masterlabpro_benchmark tests/benchmark.cpp)
target_include_directories(masterlabpro_benchmark PRIVATE src)
if(MSVC)
  target_compile_options(masterlabpro_benchmark PRIVATE /W4 /WX)
  target_link_libraries(masterlabpro_benchmark PRIVATE psapi)
else()
  target_compile_options(masterlabpro_benchmark PRIVATE -Wall -Wextra -Werror)
endif()

option(MASTERLABPRO_GUI_CHECK "Compile JUCE GUI components" OFF)
option(MASTERLABPRO_CONSOLE "Build standalone development GUI" OFF)
if(MASTERLABPRO_GUI_CHECK OR MASTERLABPRO_CONSOLE)
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
if(MASTERLABPRO_GUI_CHECK)
  add_executable(masterlabpro_gui_compile tests/gui_compile.cpp)
  target_include_directories(masterlabpro_gui_compile PRIVATE src)
  target_link_libraries(masterlabpro_gui_compile PRIVATE juce::juce_audio_processors juce::juce_gui_basics)
  target_compile_definitions(masterlabpro_gui_compile PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  add_test(NAME gui_compile COMMAND masterlabpro_gui_compile)
  add_executable(masterlabpro_gui_interaction tests/gui_interaction.cpp)
  target_include_directories(masterlabpro_gui_interaction PRIVATE src)
  target_link_libraries(masterlabpro_gui_interaction PRIVATE juce::juce_audio_processors juce::juce_gui_basics)
  target_compile_definitions(masterlabpro_gui_interaction PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  add_test(NAME gui_interaction COMMAND masterlabpro_gui_interaction)
  set(MASTERLABPRO_ASSET_ROOT "" CACHE PATH "Approved A+C assets directory for strict GUI render")
  if(WIN32)
    add_executable(masterlabpro_gui_render tests/gui_render.cpp)
    target_include_directories(masterlabpro_gui_render PRIVATE src)
    target_link_libraries(masterlabpro_gui_render PRIVATE juce::juce_audio_processors juce::juce_gui_basics)
    target_compile_definitions(masterlabpro_gui_render PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    add_test(NAME gui_render COMMAND masterlabpro_gui_render rack-smoke.png)
    if(MASTERLABPRO_ASSET_ROOT)
      add_test(NAME gui_render_ac COMMAND masterlabpro_gui_render rack-ac.png "${MASTERLABPRO_ASSET_ROOT}")
      add_test(NAME gui_render_ac_empty COMMAND masterlabpro_gui_render rack-ac-empty-100.png "${MASTERLABPRO_ASSET_ROOT}" empty 1.0)
      add_test(NAME gui_render_ac_ten COMMAND masterlabpro_gui_render rack-ac-ten-100.png "${MASTERLABPRO_ASSET_ROOT}" ten 1.0)
      add_test(NAME gui_render_ac_global COMMAND masterlabpro_gui_render rack-ac-global-100.png "${MASTERLABPRO_ASSET_ROOT}" signal-global 1.0)
      add_test(NAME gui_render_ac_module COMMAND masterlabpro_gui_render rack-ac-module-150.png "${MASTERLABPRO_ASSET_ROOT}" signal-module 1.5)
      add_test(NAME gui_render_ac_delta COMMAND masterlabpro_gui_render rack-ac-delta-200.png "${MASTERLABPRO_ASSET_ROOT}" signal-delta 2.0)
    endif()
  endif()
endif()

if(MASTERLABPRO_CONSOLE)
  juce_add_gui_app(MasterLabProConsole PRODUCT_NAME "Jerzy MasterLabPro GUI")
  target_sources(MasterLabProConsole PRIVATE src/ConsoleMain.cpp)
  target_include_directories(MasterLabProConsole PRIVATE src)
  target_link_libraries(MasterLabProConsole PRIVATE juce::juce_gui_extra juce::juce_audio_utils juce::juce_audio_processors)
  target_compile_definitions(MasterLabProConsole PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
endif()
